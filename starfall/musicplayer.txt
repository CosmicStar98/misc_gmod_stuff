require("./lib/common.txt") -- Load the dependencies

local CMDS = util.CommandWorker({ prefix = "?" })

local radioCFG = {
    global_aud = 0,
    playlist = "default",  -- enter "default" for standard playlist or a url for a remote playlist

    defaults = {
        pitch = 0,
        volume = 1.3,
        radius = 150
    }
}


-- [START] do not modify!!!

-- Data for various commands
local song_length = nil
local curr_pitch = radioCFG.defaults["pitch"]
local curr_volume = radioCFG.defaults["volume"]
local curr_radius = radioCFG.defaults["radius"]

local i = 1
local num_id = 1

local isPaused = 0

local SongList = require("./lib/musicplayer_songlist.txt") -- Default song playlist

-- [END] do not modify!!!


if SERVER then
    CMDS:registerCommand( "next", function( ply, num )
        local inputVal = tonumber(num) or 1

        net.start("skipSong")
          net.writeDouble(inputVal)
        net.send()
    end, {"n", "skip"} )

    CMDS:registerCommand( "prev", function( ply, num )
        local inputVal = tonumber(num) or 1

        net.start("prevSong")
          net.writeDouble(inputVal)
        net.send()
    end, {"b", "back"} )

    CMDS:registerCommand( "load", function( ply, ID )
        if ID then
            net.start("loadID")
              net.writeDouble(tonumber(ID))
            net.send()
        end
    end, {"play"} )

    CMDS:registerCommand( "time", function( ply, time )
        if time:sub(1, 1) == "-" or time:sub(1, 1) == "+" then
            inputType = time:sub(1, 1)
            inputVal = tonumber(time:sub(2))
        else
            inputType = ""
            inputVal = tonumber(time)
        end

        if !time then
            inputType = ""
            inputVal = 10
        end

        net.start("songTime")
          net.writeString(inputType)
          net.writeDouble(inputVal)
        net.send()
    end, {"t"} )

    CMDS:registerCommand( "pitch", function( ply, pitch )
        local inputVal = tonumber(pitch) or radioCFG.defaults["pitch"]

        net.start("songPitch")
          net.writeDouble(inputVal)
          curr_pitch = inputVal
        net.send()
    end, {"s", "speed"} )

    CMDS:registerCommand( "volume", function( ply, vol )
        local inputVal = tonumber(vol) or radioCFG.defaults["volume"]

        net.start("songVolume")
          net.writeDouble(inputVal)
          curr_volume = inputVal
        net.send()
    end, {"v", "vol"} )

    CMDS:registerCommand( "radius", function( ply, radius )
        local inputVal = tonumber(radius) or radioCFG.defaults["radius"]

        if radioCFG["global_aud"] == 0 then
            net.start("songRadius")
              net.writeDouble(inputVal)
              curr_radius = inputVal
            net.send()
        end
    end, {"r", "d", "range", "dist"} )

    CMDS:registerCommand( "pause", function( ply )
        net.start("pauseSong")
        net.send()
    end, {"p"} )
    CMDS:registerCommand( "unpause", function( ply )
        net.start("force_unpauseSong")
        net.send()
    end, {"up"} )

    CMDS:registerCommand( "global", function( ply )
        net.start("songGlobal")
        net.send()
    end, {"g", "no3d"} )

    CMDS:registerCommand( "reload", function( ply )
        net.start("reload")
        net.send()
    end, {"r", "restart", "refresh"} )

    CMDS:registerCommand( "loop", function( ply )
        net.start("loop")
        net.send()
    end, {"repeat"} )


    hook.add( "think", "ChipTweaks", function()
        if isValid(chip()) then
            chip():enableMotion(false)
            chip():setDrawShadow(false)
        end
    end )
    hook.add( "PhysgunPickup", "AllowPickup", function( ply, ent )
        if isValid(chip()) then return false end
    end )
    hook.add( "EntityTakeDamage", "NoChipDamage", function()
        if isValid(chip()) then return false end
    end )
    --[[ hook.add( "OnPhysgunPickup", "FuckOff", function( ply, ent )
        if isValid(chip()) and ply ~= owner() then
            ply:stripWeapons()
        end
    end ) ]]
else
    local function loadSong( songURL )
        bass.loadURL(tostring(songURL), "3d noblock", function( snd, err, errtxt )
            if snd then
                pcall(snd.setLooping, snd, true)
            end

            if isValid(snd) then
                snd:setVolume(curr_volume)
                snd:setFade(curr_radius, curr_radius)
            end

            -- Flush the previously created timers on each execution to avoid hitting the timer limit
            if timer.exists("startstop") then timer.remove("startstop") end
            if timer.exists("loadsong") then timer.remove("loadsong") end

            if isValid(snd) then
                local len = snd:getLength()
                local dur = snd:getTime()

                song_length = len

                setName(string.format("♪ Super Boombox DX ♪ ]\n[   %s by %s   ]\n[   ID #%i   ]\n[  %i Songs Loaded ", SongList[i]["name"], SongList[i]["artist"], i, #SongList))

                if i > #SongList then i = 1 end
                if len == nil then
                    if i >= #SongList then i = 1 else i = i + 1 end
                    loadSong(SongList[i]["url"])
                end

                -- Recreate the timers only if the audio is valid
                timer.create("startstop", len, 1, function()
                    if snd then snd:stop() end
                end)
                timer.create("loadsong", len, 1, function()
                    if i >= #SongList then i = 1 else i = i + 1 end
                    loadSong(SongList[i]["url"])
                end)

                net.receive( "skipSong", function()
                    local num_next = net.readDouble()
                    if num_next == nil or num_next <= 0 then return "" end
                    if num_next > 15 then num_next = 15 end

                    timer.stop("startstop")
                    timer.stop("loadsong")

                    timer.adjust("startstop", len, 1, function()
                        if snd then snd:stop() end
                    end)

                    timer.adjust("loadsong", len, 1, function()
                        if i >= #SongList then
                            i = 1
                        else
                            i = i + 1
                        end
                        loadSong(SongList[i]["url"])
                    end)

                    if snd then snd:stop() end

                    i = i + num_next
                    if i > #SongList then
                        i = ((i - 1) % #SongList) + 1
                    end

                    loadSong(SongList[i]["url"])
                end )

                net.receive( "prevSong", function()
                    local num_prev = net.readDouble()
                    if num_prev < 1 then return "" end
                    if num_prev > 15 then num_prev = 15 end
                    if num_prev == nil then num_prev = 1 end

                    timer.stop("startstop")
                    timer.stop("loadsong")

                    timer.adjust("startstop", len, 1, function()
                        if snd then snd:stop() end
                    end)
                    timer.adjust("loadsong", len, 1, function()
                        if i >= #SongList then i = 1 else i = i + 1 end
                        loadSong(SongList[i]["url"])
                    end)
                    
                    if snd then snd:stop() end

                    i = i - num_prev

                    if i < 1 then
                        i = #SongList + i
                    end

                    loadSong(SongList[i]["url"])
                end )

                net.receive( "loadID", function()
                    num_id = net.readDouble()
                    if num_id > #SongList then num_id = 1 end
                    if num_id < 1 then num_id = #SongList end
                    if num_id == nil then num_id = 1 end

                    timer.stop("startstop")
                    timer.stop("loadsong")
                        timer.adjust("startstop", len, 1, function()
                            if snd then snd:stop() end
                        end)
                        timer.adjust("loadsong", len, 1, function()
                            if i >= #SongList then i = 1 else i = i + 1 end
                            loadSong(SongList[i]["url"])
                        end)
                    if snd then snd:stop() end
                    i = num_id

                    loadSong(SongList[i]["url"])
                end )

                net.receive( "pauseSong", function()
                    isPaused = isPaused == 1 and 0 or 1

                    if isPaused == 1 then
                        if snd then snd:pause() end
                        timer.pause("startstop")
                        timer.pause("loadsong")
                    else
                        if snd then snd:play() end
                        timer.("startstop")
                        timer.("loadsong")
                    end
                end )

                net.receive( "force_unpauseSong", function()
                    if isPaused == 1 then
                        if snd then snd:play() end
                        timer.unpause("startstop")
                        timer.unpause("loadsong")
                    end
                end )

                net.receive( "songTime", function()
                    local type, time = net.readString(), net.readDouble()
                    if time <= 0 then time = 0 end
                    if time >= math.floor(math.abs(len)) then time = 0 end

                    if type == "+" then
                        if snd:getTime() + time >= math.floor(math.abs(len)) then
                            time = 0
                        else time = snd:getTime() + time end
                    elseif type == "-" then
                        if snd:getTime() - time < 0 then
                            time = snd:getTime()
                        else time = snd:getTime() - time end
                    else time = net.readDouble() end

                    snd:setTime(time, false)
                end )

                net.receive( "songPitch", function()
                    local pitch = net.readDouble()
                    if pitch == nil or !pitch then pitch = radioCFG.defaults["pitch"] end

                    print("\n♫ Pitch changed to "..pitch.."!")
                    curr_pitch = pitch
                    if snd then snd:setPitch(pitch) end
                end )

                net.receive( "songVolume", function()
                    local volume = net.readDouble()
                    if volume == nil or !volume then volume = radioCFG.defaults["volume"] end

                    print("\n♫ Volume changed to "..volume.."!")
                    curr_volume = volume
                    if snd then snd:setVolume(volume) end
                end )

                if radioCFG["global_aud"] == 0 then
                    net.receive( "songRadius", function()
                        local radius = net.readDouble()
                        if radius == nil or !radius then radius = radioCFG.defaults["radius"] end

                        print("\n⚠︎ Radius changed to "..radius.."!")
                        curr_radius = radius
                        if snd then snd:setFade(radius, radius) end
                    end )
                end

                net.receive( "songGlobal", function()
                    if radioCFG["global_aud"] == 1 then radioCFG["global_aud"] = 0 else radioCFG["global_aud"] = 1 end
                end )

                net.receive( "reload", function()
                    print("\n⚠︎ Reloaded the chip!")

                    if timer.exists("startstop") then timer.remove("startstop") end
                    if timer.exists("loadsong") then timer.remove("loadsong") end

                    i = i
                    num_id = i
                    if snd then snd:stop() end

                    loadSong(SongList[i]["url"])
                end )

                net.receive( "loop", function()
                    print("\n♫ Looping #"..i.."!")

                    if timer.exists("startstop") then
                        timer.remove("startstop")
                    end
                    if timer.exists("loadsong") then
                        timer.remove("loadsong")
                    end

                    timer.create("startstop", len, 0, function() snd:setTime(0, false) end)
                    timer.adjust("startstop", len, 1, function()
                        i = i
                        snd:setTime(0, false)
                    end)
                    timer.create("loadsong", len, 0, function() snd:setTime(0, false) end)
                    timer.adjust("loadsong", len, 1, function()
                        i = i
                        snd:setTime(0, false)
                    end)

                    i = i

                    snd:setTime(0, false)
                end )

                local FADE_START = 0.5  -- fade begins at 50% of the radius (0 = fade over the whole radius, 1 = hard cutoff)
                
                hook.add( "think", "snd", function()
                    if !( snd and snd:isValid() ) then return end
                    if !( isValid( chip() ) and isValid( player() ) ) then return end
                
                    if radioCFG["global_aud"] == 0 then
                        snd:setPos( chip():getPos() )
                
                        local dist = player():getPos():getDistance( chip():getPos() )
                        local fadeFrom = curr_radius * FADE_START
                        local fadeLen = math.max( curr_radius - fadeFrom, 1 )
                
                        -- 0 inside fadeFrom, ramps to 1 at curr_radius
                        local t = math.clamp( ( dist - fadeFrom ) / fadeLen, 0, 1 )
                
                        snd:setVolume( curr_volume * ( 1 - t ) )
                    else
                        snd:setPos( player():getPos() )
                        snd:setVolume( curr_volume )
                    end
                end )
            else print("🛈 "..errtxt..": "..err) end
        end)

        url = nil
    end

    print([[
     ⛧ 『 Welcome to the radio! 』 ⛧
      ♦ Heres the list of available commands:

    - ?load <NUMBER>  ::  Jumps to a song id in the playlist
    - ?back, ?next <NUMBER>  ::  FF or RW the curr song
    - ?vol <NUMBER>  ::  Changes the radio's curr volume
    - ?pitch <NUMBER>  ::  Changes the speed of the audio
    - ?radius <NUMBER>  ::  Changes the max audio distance
    - ?global <TOGGLE>  ::  Play across the map regardless of radius
    - ?time -/+<NUMBER> ::  Skips to a point in a song
    - ?loop <TOGGLE>  ::  Loops the current song until you change tracks
    - ?pause <TOGGLE>  ::  Pauses / Unpauses a track
    - ?unpause  ::  Forcefully npauses a track in case of lag
    - ?reload  ::  Remotely reboots the radio chip
    ]])


    if song_length == nil then
        if ( radioCFG["playlist"]:lower() == "default" ) then
                SongList = DefaultPlaylist
                loadSong(SongList[1]["url"])
                print("\n🛈 Loaded default playlist!")
        else
            if http.canRequest() then
                local playlist = radioCFG["playlist"]

                local function onSuccess(response, success)
                    SongList = convertToTable(response)
                    loadSong(SongList[1]["url"])
                    print("\n🛈 Loaded custom playlist!")
                end

                local function onFailure(error)
                    print(error)
                    SongList = DefaultPlaylist
                    print("\n🛈 Loaded default playlist!")
                end

                if string.startWith(playlist, "http") then
                    playlist = FixPlaylistUrl(playlist)
                end

                http.get(playlist:gsub("^https?://", ""), onSuccess, onFailure, {["Content-Type"] = "text/*"})
            else print("🛈 Unable to load custom playlist, falling back to default: Disallowed") end
        end
    end

    --[[ setupPermissionRequest({"bass.loadURL"}, "Let the radio use mp3 files from external sites", true)

    hook.add( "permissionrequest", "permission", function()
        if hasPermission("bass.loadURL", url) then
            loadSong(url)
        else print("🛈 Permission Denied") end
    end ) ]]
end
