--[[
    Like a Lua table, a Stack is a container. It follows the principle of LIFO (last in, first out).
    The Stack works like a stack of papers: the first page you put down (push) will be the last one you remove (pop).
    That also means that the last page you put down, will be the first to be removed.


    This was ripped from gmod's lua because I wanted to use it and starfall didn't have it...

    EXAMPLE USAGE:
        local NewStack = util.Stack(); -- Creating a Stack

        NewStack:Push("Hello World!") -- Push "Hello World!" onto the stack
        NewStack:Push(print) -- Push the print function onto the Stack

        local PrintFunc = NewStack:Top() -- Get the print Function
        NewStack:Pop() -- Remove the print Function from the stack
        PrintFunc(NewStack:Top()) -- print "Hello World!"
        NewStack:Pop() -- Remove "Hello World!"
]]
local function PopStack( self, num )
if ( num == nil ) then
        num = 1
    elseif ( num < 0 ) then
        error( string.format( "attempted to pop %d elements in stack, expected >= 0", num ), 3 )
    else
        num = math.floor( num )
    end

    local len = self[ 0 ]

    if ( num > len ) then
        error( string.format( "attempted to pop %u element%s in stack of length %u", num, num == 1 and "" or "s", len ), 3 )
    end

    return num, len

end

local STACK = {
    Push = function( self, obj )
        local len = self[ 0 ] + 1
        self[ len ] = obj
        self[ 0 ] = len
    end,

    Pop = function( self, num )
        local len
        num, len = PopStack( self, num )

        if ( num == 0 ) then
            return nil
        end

        local newlen = len - num
        self[ 0 ] = newlen

        newlen = newlen + 1
        local ret = self[ newlen ]

        -- Pop up to the last element
        for i = len, newlen, -1 do
            self[ i ] = nil
        end

        return ret
    end,

    PopMulti = function( self, num )
        local len
        num, len = PopStack( self, num )

        if ( num == 0 ) then
            return {}
        end

        local newlen = len - num
        self[ 0 ] = newlen

        local ret = {}
        local retpos = 0

        -- Pop each element and add it to the table
        -- Iterate in reverse since the stack is internally stored
        -- with 1 being the bottom element and len being the top
        -- But the return will have 1 as the top element
        for i = len, newlen + 1, -1 do
            retpos = retpos + 1
            ret[ retpos ] = self[ i ]

            self[ i ] = nil
        end

        return ret
    end,

    Top = function( self )
        local len = self[ 0 ]

        if ( len == 0 ) then
            return nil
        end

        return self[ len ]
    end,

    Size = function( self )
        return self[ 0 ]
    end
}

STACK.__index = STACK

util = util or {}
function util.Stack()
    return setmetatable( { [ 0 ] = 0 }, STACK )
end


--[[
    Chat command system inspired by ULib
    TODO: find a way to make this data be shared between different chips which would use this file.
    TODO: add usage hints

    EXAMPLE USAGE:
        -- this must be called before using anything. inside, you can define custom configurations here. (eg. a prefix)
        local CMD = util.CommandWorker({
            prefix    = "!",                               -- default >
            ownerOnly = true,                              -- default true, see ACCESS RULES below
            whitelist = { "STEAM_0:1:1234567", "76561198000000000" }, -- SteamIDs on whistlist per worker (this is on top of the global default whitelist; if specified)
            useDefaultWhitelist = true,                    -- default true: also honor DEFAULT_WHITELIST below
        })
        CMD:registerCommand("example_cmd", function( ply, any_param_name ) -- command name & function
            local inputVal = any_param_name -- careful, the type for param names are strings by default! they must be converted afterwards!

            net.start("something")
            net.send()
        end, {"alias1", "alias2"}) -- Aliases are nicknames for the actual command you're registering


    ACCESS RULES:
        - The chip owner is always allowed.
        - Whitelisted players (DEFAULT_WHITELIST + config.whitelist) are always allowed.
        - Everyone else is allowed ONLY if ownerOnly == false and the whitelist is empty.
          (so ownerOnly = true  -> owner + whitelist
             ownerOnly = false + non-empty whitelist -> owner + whitelist
             ownerOnly = false + empty whitelist     -> everyone)
]]
local COMMANDS = {
    registerCommand = function( self, name, func, aliasList )
        name = name:lower()
        local key = self.prefix .. name

        -- instead of throwing a fit, append a number
        if ( self.commands[key] ~= nil ) then
            self.conflicts = self.conflicts + 1
            key = key .. "_" .. self.conflicts

            print( Color(255, 20, 20), "[ERROR] :: Command '" .. name .. "' for prefix '" .. self.prefix .. "' already registered!\n" )
            print( Color(255, 174, 23), "[WARN] :: Command '" .. name .. "' for prefix '" .. self.prefix .. "' has been given a temp name registration! (use '".. key .. ")'" )
        end

        if ( type(func) ~= "function" ) then
            print( Color(255, 20, 20), "[ERROR] :: Command '" .. name .. "' for prefix '" .. self.prefix .. "' must contain a function!\n" )
            return error("Command '" .. name .. "' for prefix '" .. self.prefix .. "' must contain a function!")
        end

        self.commands[key] = func

        if ( aliasList ) then -- TODO: program feature to allow aliases to extend the original command to allow for additional features per alias (Low priority)
            for _, alias in ipairs(aliasList) do
                alias = alias:lower()
                if ( alias ~= name ) then
                    self.aliases[self.prefix .. alias] = key
                end
            end
        end

        return self
    end,

    getPrefix = function( self )
        return self.prefix
    end,

    addWhitelist = function( self, id )
        self.whitelist[tostring(id)] = true
        return self
    end,

    removeWhitelist = function( self, id )
        self.whitelist[tostring(id)] = nil
        return self
    end,

    isWhitelisted = function( self, target )
        if ( type(target) == "string" or type(target) == "number" ) then
            return self.whitelist[tostring(target)] == true
        end

        if !isValid( target ) then return false end

        return self.whitelist[target:getSteamID()] == true
            or self.whitelist[ tostring(target:getSteamID64()) ] == true
    end,

    isAuthorized = function( self, ply )
        if !isValid( ply ) then return false end
        if ply == owner() then return true end
        if self:isWhitelisted(ply) then return true end

        -- open access only when not owner-only and no whitelist is in use
        return ( !self.ownerOnly ) and next( self.whitelist ) == nil
    end
}

COMMANDS.__index = COMMANDS


local workerCount = 0
-- Default whitelist, applied to every worker unless it sets useDefaultWhitelist = false.
local DEFAULT_WHITELIST = {
    -- "STEAM_0:1:1234567",
    -- "76561198000000000",
}
function util.CommandWorker( config )
    config = config or {}
    workerCount = workerCount + 1

    local self = setmetatable( {
        prefix    = config.prefix or ">",
        ownerOnly = config.ownerOnly ~= false, -- defaults to true
        whitelist = {},
        commands  = {},
        aliases   = {},
        conflicts = 0,
        hookName  = "CommandWorker_" .. workerCount,
    }, COMMANDS )

    if ( config.useDefaultWhitelist ~= false ) then
        for _, id in ipairs(DEFAULT_WHITELIST) do
            self:addWhitelist(id)
        end
    end

    if ( config.whitelist ) then
        for _, id in ipairs(config.whitelist) do
            self:addWhitelist(id)
        end
    end

    hook.add( "PlayerSay", self.hookName, function( ply, text, teamChat )
        if !self:isAuthorized(ply) then return end

        local args = string.explode(" ", text)
        local cmd = args[1]:lower()
        cmd = self.aliases[cmd] or cmd

        local func = self.commands[cmd]
        if ( func ) then
            func( ply, unpack(args, 2) ) -- skip args[1], the command itself
            return "" -- hide the command from chat
        end
    end )

    hook.add( "removed", self.hookName .. "_cleanup", function()
        hook.remove("PlayerSay", self.hookName)
        self.commands, self.aliases = {}, {}
    end )


    -- < Default chat commands > -------------------------------------------
    -- TODO: move this into a table of some kind and use a for-loop to automatically register them all (low priority until the list of built-in commands is long enough to justify)

    -- lists every command with its aliases
    -- TODO: add optional command hints & descriptions
    self:registerCommand( "help", function( ply )
        local keys = {}
        for key in pairs(self.commands) do keys[#keys + 1] = key end
        table.sort(keys)

        for _, key in ipairs(keys) do
            local names = { key }
            for alias, target in pairs(self.aliases) do
                if ( target == key ) then names[#names + 1] = alias end
            end

            print( Color(120, 200, 255), table.concat( names, " | " ) )
        end
    end )

    -- pointless until the above TODO is implemented
    --[[ if isFirstTimePredicted() then
        print(Color(39, 149, 245), "[INFO] :: Chip successfully loaded!\nType ".. (config.prefix or ">") .. "help for a list of available commands.")
    end ]]

    return self
end


-- ripped from pac3 code
function FixUrl(url)
    url = url:Trim():gsub("[\"'<>\n\\]+", "")

    if url:find("dropbox", 1, true) then
        url = url:gsub([[^http%://dl%.dropboxusercontent%.com/]], [[https://dl.dropboxusercontent.com/]])
        url = url:gsub([[^https?://dl.dropbox.com/]], [[https://www.dropbox.com/]])
        url = url:gsub([[^https?://www.dropbox.com/s/(.+)%?dl%=[01]$]], [[https://dl.dropboxusercontent.com/s/%1]])
        url = url:gsub([[^https?://www.dropbox.com/s/(.+)$]], [[https://dl.dropboxusercontent.com/s/%1]])
        url = url:gsub([[^https?://www.dropbox.com/scl/(.+)$]], [[https://dl.dropboxusercontent.com/scl/%1]]) -- Fix for new dropbox format.
        return url
    end

    if url:find("drive.google.com", 1, true) and not url:find("export=download", 1, true) then
        local id =
            url:match("https://drive.google.com/file/d/(.-)/") or
            url:match("https://drive.google.com/file/d/(.-)$") or
            url:match("https://drive.google.com/open%?id=(.-)$")

        if id then
            return "https://drive.google.com/uc?export=download&id=" .. id
        end
        return url
    end

    if url:find("gitlab.com", 1, true) then
        return url:gsub("^(https?://.-/.-/.-/)blob", "%1raw")
    end

    url = url:gsub([[^http%://onedrive%.live%.com/redir?]],[[https://onedrive.live.com/download?]])
    url = url:gsub("pastebin.com/([a-zA-Z0-9]*)$", "pastebin.com/raw/%1")
    -- url = url:gsub("github.com/([a-zA-Z0-9_]+)/([a-zA-Z0-9_]+)/blob/", "github.com/%1/%2/raw/")

    return url
end

-- ripped from mediaplayer code
function FixPlaylistUrl(url)
    url = url:Trim():gsub("[\"'<>\n\\]+", "")


    if url:find("gitlab.com", 1, true) then
        return url:gsub("^(https?://.-/.-/.-/)blob", "%1raw")
    end

    url = url:gsub("pastebin.com/([a-zA-Z0-9]*)$", "pastebin.com/raw/%1")

    return url
end


-- Playlist parser
-- Example input:
-- >Example1|Example2|Example3,Example4|Example5|Example6,Example7|Example8|Example9<
-- The required order to follow is: song's name, author, link to file
-- CASE SENSITIVE. CHECK YOUR SHIT BEFORE COMPLAINING YOUR PLAYLIST DOESN'T WORK
-- LARGE PLAYLISTS MAY TAKE LONGER TO PROCESS!!
function convertToTable(input)
    local result = {}
    --local currentKey = -1
    local currentKey = 0
    local currentEntry = {}
    input = input:gsub("[<>]", "")

    local keys = {}
    for key in input:gmatch("[^,]+") do
        table.insert(keys, key)
    end

    for _, key in ipairs(keys) do
        local entries = {}
        for entry in key:gmatch("[^|]+") do
            table.insert(entries, entry)
        end

        currentKey = currentKey + 1
        currentEntry = {}
        currentEntry["name"] = entries[1] or "Unknown"
        currentEntry["artist"] = entries[2] or "Unknown"
        currentEntry["url"] = FixUrl(entries[3]) or "Unknown"

        result[currentKey] = currentEntry
    end

    return result
end
